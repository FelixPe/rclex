  if (message_p->dim.size > message_p->dim.capacity)
    return raise_with_message(env, __FILE__, __LINE__, "invalid sequence size/capacity");

  if (message_p->dim.size > 0 && message_p->dim.data == NULL)
    return raise_with_message(env, __FILE__, __LINE__, "NULL sequence data with non-zero size");


  ERL_NIF_TERM dim[(message_p->dim.size > 0 ? message_p->dim.size : 1)];

  for (size_t dim_i = 0; dim_i < message_p->dim.size; ++dim_i)
  {
    ERL_NIF_TERM dim_label_term = enif_make_binary_wrapper(env, message_p->dim.data[dim_i].label.data, message_p->dim.data[dim_i].label.size);
    if (enif_is_exception(env, dim_label_term))
      return dim_label_term;
    dim[dim_i] = enif_make_tuple(env, 3,
      dim_label_term,
      enif_make_uint(env, message_p->dim.data[dim_i].size),
      enif_make_uint(env, message_p->dim.data[dim_i].stride)
    );
  }

  return enif_make_tuple(env, 2,
    (message_p->dim.size == 0 ? enif_make_list(env, 0) : enif_make_list_from_array(env, dim, message_p->dim.size)),
    enif_make_uint(env, message_p->data_offset)
  );
