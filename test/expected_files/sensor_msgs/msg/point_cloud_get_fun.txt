  if (message_p->points.size > message_p->points.capacity)
    return raise_with_message(env, __FILE__, __LINE__, "invalid sequence size/capacity");

  if (message_p->points.size > 0 && message_p->points.data == NULL)
    return raise_with_message(env, __FILE__, __LINE__, "NULL sequence data with non-zero size");


  ERL_NIF_TERM points[(message_p->points.size > 0 ? message_p->points.size : 1)];

  for (size_t points_i = 0; points_i < message_p->points.size; ++points_i)
  {
    points[points_i] = enif_make_tuple(env, 3,
      enif_make_double(env, message_p->points.data[points_i].x),
      enif_make_double(env, message_p->points.data[points_i].y),
      enif_make_double(env, message_p->points.data[points_i].z)
    );
  }

  if (message_p->channels.size > message_p->channels.capacity)
    return raise_with_message(env, __FILE__, __LINE__, "invalid sequence size/capacity");

  if (message_p->channels.size > 0 && message_p->channels.data == NULL)
    return raise_with_message(env, __FILE__, __LINE__, "NULL sequence data with non-zero size");


  ERL_NIF_TERM channels[(message_p->channels.size > 0 ? message_p->channels.size : 1)];

  for (size_t channels_i = 0; channels_i < message_p->channels.size; ++channels_i)
  {
    if (message_p->channels.data[channels_i].values.size > message_p->channels.data[channels_i].values.capacity)
      return raise_with_message(env, __FILE__, __LINE__, "invalid sequence size/capacity");

    if (message_p->channels.data[channels_i].values.size > 0 && message_p->channels.data[channels_i].values.data == NULL)
      return raise_with_message(env, __FILE__, __LINE__, "NULL sequence data with non-zero size");


    ERL_NIF_TERM channels_values[(message_p->channels.data[channels_i].values.size > 0 ? message_p->channels.data[channels_i].values.size : 1)];

    for (size_t channels_values_i = 0; channels_values_i < message_p->channels.data[channels_i].values.size; ++channels_values_i)
    {
      channels_values[channels_values_i] = enif_make_double(env, message_p->channels.data[channels_i].values.data[channels_values_i]);
    }

    ERL_NIF_TERM channels_name_term = enif_make_binary_wrapper(env, message_p->channels.data[channels_i].name.data, message_p->channels.data[channels_i].name.size);
    if (enif_is_exception(env, channels_name_term))
      return channels_name_term;
    channels[channels_i] = enif_make_tuple(env, 2,
      channels_name_term,
      (message_p->channels.data[channels_i].values.size == 0 ? enif_make_list(env, 0) : enif_make_list_from_array(env, channels_values, message_p->channels.data[channels_i].values.size))
    );
  }

  ERL_NIF_TERM header_frame_id_term = enif_make_binary_wrapper(env, message_p->header.frame_id.data, message_p->header.frame_id.size);
  if (enif_is_exception(env, header_frame_id_term))
    return header_frame_id_term;
  return enif_make_tuple(env, 3,
    enif_make_tuple(env, 2,
      enif_make_tuple(env, 2,
        enif_make_int(env, message_p->header.stamp.sec),
        enif_make_uint(env, message_p->header.stamp.nanosec)
      ),
      header_frame_id_term
    ),
    (message_p->points.size == 0 ? enif_make_list(env, 0) : enif_make_list_from_array(env, points, message_p->points.size)),
    (message_p->channels.size == 0 ? enif_make_list(env, 0) : enif_make_list_from_array(env, channels, message_p->channels.size))
  );
